cmake_minimum_required(VERSION 3.16)
project(lsm9ds1_linux_wrapper VERSION 1.0.0 LANGUAGES C CXX)

option(LSM9DS1_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

add_library(lsm9ds1 src/lsm9ds1.c)
target_include_directories(lsm9ds1 PUBLIC include)
target_compile_features(lsm9ds1 PUBLIC c_std_11)
target_compile_options(lsm9ds1 PRIVATE -Wall -Wextra -Wpedantic)
if(LSM9DS1_WARNINGS_AS_ERRORS)
  target_compile_options(lsm9ds1 PRIVATE -Werror)
endif()
target_link_libraries(lsm9ds1 PUBLIC m)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(lsm9ds1_linux src/linux_i2c.c)
  target_include_directories(lsm9ds1_linux PUBLIC include)
  target_compile_features(lsm9ds1_linux PUBLIC c_std_11)
  target_compile_options(lsm9ds1_linux PRIVATE -Wall -Wextra -Wpedantic)
  if(LSM9DS1_WARNINGS_AS_ERRORS)
    target_compile_options(lsm9ds1_linux PRIVATE -Werror)
  endif()
  target_link_libraries(lsm9ds1_linux PUBLIC lsm9ds1)

  add_executable(lsm9ds1-cli src/cli.c)
  target_compile_options(lsm9ds1-cli PRIVATE -Wall -Wextra -Wpedantic)
  if(LSM9DS1_WARNINGS_AS_ERRORS)
    target_compile_options(lsm9ds1-cli PRIVATE -Werror)
  endif()
  target_link_libraries(lsm9ds1-cli PRIVATE lsm9ds1_linux)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(lsm9ds1_tests tests/test_lsm9ds1.cpp)
  target_compile_features(lsm9ds1_tests PRIVATE cxx_std_17)
  target_compile_options(lsm9ds1_tests PRIVATE -Wall -Wextra -Wpedantic)
  if(LSM9DS1_WARNINGS_AS_ERRORS)
    target_compile_options(lsm9ds1_tests PRIVATE -Werror)
  endif()
  target_link_libraries(lsm9ds1_tests PRIVATE lsm9ds1)
  add_test(NAME lsm9ds1_tests COMMAND lsm9ds1_tests)
endif()
